# The storefront (home, categories, product families) is open to every crawler.
#
# Disallowed: areas behind a sign-in, per-visitor pages, and the API. Search is
# out too: every query string is a new URL, so a crawler following search links
# never runs out of pages and burns the monthly request quota doing it.
#
# Each locale is spelled out rather than written `/*/rep`: in robots.txt `*`
# also matches slashes, so the wildcard would hide any product family whose
# slug starts with "rep", "cart", "quote" and so on (/fa/f/repair-kits).
#
# Public pages that must stay out of results (/track, /pay, /invoice) are NOT
# listed on purpose: they carry a `noindex` tag, and a crawler barred here
# never fetches the page to see it, so it could still list the bare URL.
User-agent: *
Disallow: /api/
Disallow: /en/admin
Disallow: /fa/admin
Disallow: /en/rep
Disallow: /fa/rep
Disallow: /en/account
Disallow: /fa/account
Disallow: /en/cart
Disallow: /fa/cart
Disallow: /en/quick-order
Disallow: /fa/quick-order
Disallow: /en/quote
Disallow: /fa/quote
Disallow: /en/search
Disallow: /fa/search

# OpenAI's training crawler: ~80k requests a day in October 2026, most of the
# free plan's monthly quota on its own. It only gathers training text, so
# barring it costs no visibility — ChatGPT search (OAI-SearchBot) and
# ChatGPT fetching a page on a user's behalf (ChatGPT-User) are separate
# agents, still allowed by the group above. A bot obeys only its most specific
# group, so this one is out of the whole site.
User-agent: GPTBot
Disallow: /
